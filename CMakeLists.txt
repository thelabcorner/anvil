cmake_minimum_required(VERSION 3.16)
project(anvil LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(anvil src/anvil.cpp)
if(MSVC)
  target_compile_options(anvil PRIVATE /O2 /EHsc)
else()
  target_compile_options(anvil PRIVATE -O3 -DNDEBUG -Wall -Wextra -Wpedantic)
endif()

option(ANVIL_BUILD_NATIVE_BENCH "Build Brotli/Zstd comparison benchmark" ON)
if(ANVIL_BUILD_NATIVE_BENCH)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/install/lib")
    set(_TP "${CMAKE_CURRENT_SOURCE_DIR}/third_party/install")
  endif()
  find_path(BROTLI_INCLUDE_DIR brotli/encode.h PATHS "${_TP}/include")
  find_path(ZSTD_INCLUDE_DIR zstd.h PATHS "${_TP}/include")
  find_library(BROTLIENC_LIBRARY NAMES brotlienc PATHS "${_TP}/lib")
  find_library(BROTLIDEC_LIBRARY NAMES brotlidec PATHS "${_TP}/lib")
  find_library(BROTLICOMMON_LIBRARY NAMES brotlicommon PATHS "${_TP}/lib")
  find_library(ZSTD_LIBRARY NAMES zstd zstd_static PATHS "${_TP}/lib")
  if(BROTLIENC_LIBRARY AND BROTLIDEC_LIBRARY AND BROTLICOMMON_LIBRARY AND ZSTD_LIBRARY)
    add_executable(anvil_bench tools/bench_native.cpp)
    target_include_directories(anvil_bench PRIVATE ${BROTLI_INCLUDE_DIR} ${ZSTD_INCLUDE_DIR})
    target_link_libraries(anvil_bench PRIVATE ${BROTLIENC_LIBRARY} ${BROTLIDEC_LIBRARY} ${BROTLICOMMON_LIBRARY} ${ZSTD_LIBRARY})
    if(MSVC)
      target_compile_options(anvil_bench PRIVATE /O2 /EHsc /DNDEBUG)
    else()
      target_compile_options(anvil_bench PRIVATE -O3 -DNDEBUG)
    endif()
  else()
    message(STATUS "Brotli/Zstd development libraries not found; skipping anvil_bench")
  endif()
endif()
