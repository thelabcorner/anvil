cmake_minimum_required(VERSION 3.16)
project(anvil LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/install/lib")
  set(_TP "${CMAKE_CURRENT_SOURCE_DIR}/third_party/install")
endif()
find_path(BROTLI_INCLUDE_DIR brotli/encode.h PATHS "${_TP}/include")
find_library(BROTLIENC_LIBRARY NAMES brotlienc PATHS "${_TP}/lib")
find_library(BROTLIDEC_LIBRARY NAMES brotlidec PATHS "${_TP}/lib")
find_library(BROTLICOMMON_LIBRARY NAMES brotlicommon PATHS "${_TP}/lib")

add_executable(anvil src/anvil.cpp)

# Pinned Apache-2.0 libsais primitive used by ANVIL's own BWT backend.  We
# vendor only libsais.c/libsais.h, not libbsc or its entropy/post-transform code.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libsais/src/libsais.c")
  add_library(anvil_libsais STATIC third_party/libsais/src/libsais.c)
  target_include_directories(anvil_libsais PUBLIC third_party/libsais/include)
  target_compile_definitions(anvil PRIVATE ANVIL_HAVE_LIBSAIS=1)
  target_link_libraries(anvil PRIVATE anvil_libsais)
  if(MSVC)
    target_compile_options(anvil_libsais PRIVATE /O2 /DNDEBUG)
  else()
    target_compile_options(anvil_libsais PRIVATE -O3 -DNDEBUG)
  endif()
endif()
if(BROTLI_INCLUDE_DIR AND BROTLIENC_LIBRARY AND BROTLIDEC_LIBRARY AND BROTLICOMMON_LIBRARY)
  target_compile_definitions(anvil PRIVATE ANVIL_HAVE_BROTLI=1)
  target_include_directories(anvil PRIVATE ${BROTLI_INCLUDE_DIR})
  target_link_libraries(anvil PRIVATE ${BROTLIENC_LIBRARY} ${BROTLIDEC_LIBRARY} ${BROTLICOMMON_LIBRARY})

  add_executable(brotli_lw tools/brotli_lw.cpp)
  target_include_directories(brotli_lw PRIVATE ${BROTLI_INCLUDE_DIR})
  target_link_libraries(brotli_lw PRIVATE ${BROTLIENC_LIBRARY} ${BROTLIDEC_LIBRARY} ${BROTLICOMMON_LIBRARY})
  if(MSVC)
    target_compile_options(brotli_lw PRIVATE /O2 /EHsc /DNDEBUG)
  else()
    target_compile_options(brotli_lw PRIVATE -O3 -DNDEBUG)
  endif()
endif()
if(MSVC)
  target_compile_options(anvil PRIVATE /O2 /EHsc)
  target_link_options(anvil PRIVATE /STACK:8388608)
  if(TARGET anvil_bench)
    target_link_options(anvil_bench PRIVATE /STACK:8388608)
  endif()
else()
  target_compile_options(anvil PRIVATE -O3 -DNDEBUG -Wall -Wextra -Wpedantic)
endif()

option(ANVIL_BUILD_NATIVE_BENCH "Build Brotli/Zstd comparison benchmark" ON)
if(ANVIL_BUILD_NATIVE_BENCH)
  find_path(ZSTD_INCLUDE_DIR zstd.h PATHS "${_TP}/include")
  find_library(ZSTD_LIBRARY NAMES zstd zstd_static PATHS "${_TP}/lib")
  if(BROTLIENC_LIBRARY AND BROTLIDEC_LIBRARY AND BROTLICOMMON_LIBRARY AND ZSTD_LIBRARY)
    add_executable(anvil_bench tools/bench_native.cpp)
    target_include_directories(anvil_bench PRIVATE ${BROTLI_INCLUDE_DIR} ${ZSTD_INCLUDE_DIR})
    target_link_libraries(anvil_bench PRIVATE ${BROTLIENC_LIBRARY} ${BROTLIDEC_LIBRARY} ${BROTLICOMMON_LIBRARY} ${ZSTD_LIBRARY})
    if(MSVC)
      target_compile_options(anvil_bench PRIVATE /O2 /EHsc /DNDEBUG)
    else()
      target_compile_options(anvil_bench PRIVATE -O3 -DNDEBUG)
    endif()
  else()
    message(STATUS "Brotli/Zstd development libraries not found; skipping anvil_bench")
  endif()
endif()
